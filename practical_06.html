import pandas as pd
from sklearn import model_selection
from sklearn.ensemble import AdaBoostClassifier
# Correct URL for the dataset 
url="https://raw.githubusercontent.com/jbrownlee/Datasets/master/pima-indians 
diabetes.data.csv"
names = ['preg', 'plas', 'pres', 'skin', 'test', 'mass', 'pedi', 'age', 'class'] 
dataframe = pd.read_csv(url, names=names)
array = dataframe.values 
X = array[:, 0:8]
Y = array[:, 8] 
seed = 7
num_trees = 30
# kfold makes trees with split number.
kfold = model_selection.KFold(n_splits=10, random_state=seed, shuffle=True)
# Set the algorithm to 'SAMME' to avoid deprecation warnings
model
=
AdaBoostClassifier(n_estimators=num_trees,
random_state=seed, 
algorithm='SAMME')
# cross_val_score method is used to calculate the accuracy of the model sliced into X, Y 
results = model_selection.cross_val_score(model, X, Y, cv=kfold)
print("Mean Accuracy: ", results.mean())
